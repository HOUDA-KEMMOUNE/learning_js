            online: true
        },

            username: "Sara",
            score: 1800,
            online: true
        }
    ]

Hint: You're going to discover another array method:
        .filter()

The idea is:

        players
           ↓
        filter()
           ↓
        "Keep this player?"
           ↓
        YES → keep
        NO  → remove
